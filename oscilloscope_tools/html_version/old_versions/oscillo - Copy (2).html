Why a proxy is needed + 25-line Python proxy (run once)

Browsers refuse to read responses from a server that does not send the Access-Control-Allow-Origin header (CORS). The scope does not send it, so a direct fetch() fails with "Failed to fetch" even though Python and ping work. The fix is a tiny local relay that forwards your request and adds the missing header.

One-time setup:

  1. Save the code below as proxy.py in any folder.
  2. Open a terminal: python proxy.py (Python stdlib only - no pip).
  3. Leave it running. Click the buttons above. Re-use the same window next time.
#!/usr/bin/env python3
# proxy.py - tiny local CORS proxy for the oscilloscope HTML page
# Run:  python proxy.py
# Then in the HTML enable "Use local proxy" (default ON).

import http.server, socketserver, urllib.request, urllib.parse, http.client, socket, sys

PORT = 8765   # must match "Proxy port" in the HTML

class Proxy(http.server.BaseHTTPRequestHandler):
    def _cors(self):
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET,OPTIONS')
        self.send_header('Access-Control-Allow-Headers', '*')

    def do_OPTIONS(self):
        self.send_response(204); self._cors(); self.end_headers()

    def do_GET(self):
        q = urllib.parse.urlparse(self.path).query
        target = urllib.parse.parse_qs(q).get('u', [None])[0]
        if not target:
            self.send_response(400); self._cors(); self.end_headers()
            self.wfile.write(b'missing ?u='); return
        try:
            r = urllib.request.urlopen(target, timeout=10)
            ct = r.headers.get('Content-Type', 'application/octet-stream')
            # The scope often advertises a wrong Content-Length and then closes
            # the socket early (IncompleteRead) or stops sending (read timeout).
            # In every case the bytes already received ARE the complete reply,
            # so we read in chunks and keep whatever arrives.
            chunks = []
            try:
                while True:
                    b = r.read(4096)
                    if not b:
                        break
                    chunks.append(b)
            except http.client.IncompleteRead as ir:
                chunks.append(ir.partial)
            except (socket.timeout, TimeoutError):
                pass
            data = b''.join(chunks)
            self.send_response(200)
            self.send_header('Content-Type', ct)
            self.send_header('Content-Length', str(len(data)))
            self._cors(); self.end_headers()
            self.wfile.write(data)
        except Exception as e:
            self.send_response(502); self._cors(); self.end_headers()
            try:
                self.wfile.write(str(e).encode())
            except Exception:
                pass

    def log_message(self, fmt, *args):
        sys.stderr.write("[proxy] %s - %s\n" % (self.address_string(), fmt % args))

with socketserver.ThreadingTCPServer(('127.0.0.1', PORT), Proxy) as srv:
    print(f"CORS proxy listening on http://127.0.0.1:{PORT}/   (Ctrl+C to stop)")
    srv.serve_forever()

Security note: the proxy binds 127.0.0.1 only (not exposed on the network) and only forwards GET. Stop it when you are done.
